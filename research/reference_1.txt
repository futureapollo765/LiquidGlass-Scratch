event_whenflagclicked()

DEFINE To RBG Maps []
deletealloflist(LIST=R Map)
deletealloflist(LIST=B Map)
deletealloflist(LIST=G Map)
setvariableto(VARIABLE=#colori, VALUE=0)
control_repeat(TIMES=Screen Pixel)
SUBSTACK
  changevariableby(VARIABLE=#colori, VALUE=1)
  addtolist(LIST=R Map, ITEM=(Colour Map[#colori] mod 256))
  addtolist(LIST=G Map, ITEM=(floor((Colour Map[#colori] / 256)) mod 256))
  addtolist(LIST=B Map, ITEM=(floor((Colour Map[#colori] / 65536)) mod 256))

DEFINE Render []
pen_clear()
setvariableto(VARIABLE=#id, VALUE=0)
setvariableto(VARIABLE=#y, VALUE=180)
control_repeat(TIMES=H)
SUBSTACK
  setvariableto(VARIABLE=#x, VALUE=-240)
  control_repeat(TIMES=W)
  SUBSTACK
    changevariableby(VARIABLE=#id, VALUE=1)
    control_if(CONDITION=(Distance Map[#id] < 1))
    SUBSTACK
      setvariableto(VARIABLE=#d, VALUE=Distance Map[#id])
      setvariableto(VARIABLE=#absd, VALUE=abs(#d))
      setvariableto(VARIABLE=#color id in list, VALUE=#id)
      control_if(CONDITION=(#absd < Edge distortion thickness))
      SUBSTACK
        setvariableto(VARIABLE=#d0, VALUE=Distance Map[#id])
        setvariableto(VARIABLE=#dx, VALUE=((Distance Map[(#id + 1)] - Distance Map[(#id - 1)]) / 2))
        setvariableto(VARIABLE=#dy, VALUE=((Distance Map[(#id - W)] - Distance Map[(#id + W)]) / 2))
        setvariableto(VARIABLE=#len, VALUE=sqrt(((#dx * #dx) + (#dy * #dy))))
        setvariableto(VARIABLE=#nx, VALUE=((#len > 0) * (#dx / #len)))
        setvariableto(VARIABLE=#ny, VALUE=((#len > 0) * (#dy / #len)))
        setvariableto(VARIABLE=#t, VALUE=(1 - (#absd / Edge distortion thickness)))
        setvariableto(VARIABLE=#t, VALUE=((#t > 0) * #t))
        setvariableto(VARIABLE=#strength, VALUE=(#t * (#t * #t)))
        setvariableto(VARIABLE=#sax, VALUE=round(NUM=(#x - (#nx * (#strength * Distortion)))))
        setvariableto(VARIABLE=#say, VALUE=round(NUM=(#y - (#ny * (#strength * Distortion)))))
        setvariableto(VARIABLE=#color id in list, VALUE=(((W * (180 - #say)) + (#sax + 240)) + 1))
      pen_setPenColorToColor(COLOR=(R Map[#color id in list] + ((G Map[#color id in list] * 256) + (B Map[#color id in list] * 65536))))
      control_if(CONDITION=(#absd < .5))
      SUBSTACK
        pen_changePenColorParamBy(COLOR_PARAM=pen_menu_colorParam(colorParam=brightness), VALUE=-15)
        control_if_else(CONDITION=(#ny > 0))
        SUBSTACK
          pen_changePenColorParamBy(COLOR_PARAM=pen_menu_colorParam(colorParam=brightness), VALUE=(abs(#ny) * 30))
          pen_changePenColorParamBy(COLOR_PARAM=pen_menu_colorParam(colorParam=saturation), VALUE=(abs(#ny) * -30))
        SUBSTACK2
          pen_changePenColorParamBy(COLOR_PARAM=pen_menu_colorParam(colorParam=brightness), VALUE=(abs(#ny) * 20))
          pen_changePenColorParamBy(COLOR_PARAM=pen_menu_colorParam(colorParam=saturation), VALUE=(abs(#ny) * -20))
      motion_gotoxy(X=#x, Y=#y)
      pen_penDown()
      pen_penUp()
    changevariableby(VARIABLE=#x, VALUE=1)
  changevariableby(VARIABLE=#y, VALUE=-1)

DEFINE SDF circle %s %s %s %s %s ["cx","cy","cr","x","y"]
setvariableto(VARIABLE=#dx, VALUE=($x - $cx))
setvariableto(VARIABLE=#dy, VALUE=($y - $cy))
setvariableto(VARIABLE=#distance, VALUE=(sqrt(((#dx * #dx) + (#dy * #dy))) - $cr))

DEFINE SDF rounded rectangle %s %s %s %s %s %s %s ["sx","sy","sw","sh","sr","x","y"]
setvariableto(VARIABLE=#dx, VALUE=abs(($x - $sx)))
setvariableto(VARIABLE=#dy, VALUE=abs(($y - $sy)))
setvariableto(VARIABLE=#qx, VALUE=(#dx - (($sw / 2) - $sr)))
setvariableto(VARIABLE=#qy, VALUE=(#dy - (($sh / 2) - $sr)))
setvariableto(VARIABLE=#qx2, VALUE=((#qx > 0) * #qx))
setvariableto(VARIABLE=#qy2, VALUE=((#qy > 0) * #qy))
setvariableto(VARIABLE=#outside, VALUE=sqrt(((#qx2 * #qx2) + (#qy2 * #qy2))))
setvariableto(VARIABLE=#inside, VALUE=(((#qx < #qy) * #qy) + (not(OPERAND=(#qx < #qy)) * #qx)))
setvariableto(VARIABLE=#inside, VALUE=((#inside < 0) * #inside))
setvariableto(VARIABLE=#distance, VALUE=((#outside + #inside) - $sr))

DEFINE SDF id: %s position: %s %s ["id","x","y"]
control_if(CONDITION=(#shape type[$id] = 1))
SUBSTACK
  SDF circle %s %s %s %s %s ['#shape x[$id]', '#shape y[$id]', '#shape radius[$id]', '$x', '$y']
  control_stop(STOP_OPTION=this script)
control_if(CONDITION=(#shape type[$id] = 2))
SUBSTACK
  SDF rounded rectangle %s %s %s %s %s %s %s ['#shape x[$id]', '#shape y[$id]', '#shape width[$id]', '#shape height[$id]', '#shape radius[$id]', '$x', '$y']
  control_stop(STOP_OPTION=this script)
control_if(CONDITION=(#shape type[$id] = 3))
SUBSTACK
  SDF capsule %s %s %s %s %s %s ['#shape x[$id]', '#shape y[$id]', '#shape width[$id]', '#shape height[$id]', '$x', '$y']
  control_stop(STOP_OPTION=this script)

DEFINE SDF capsule %s %s %s %s %s %s ["sx","sy","sw","sh","x","y"]
control_if(CONDITION=($sw > $sh))
SUBSTACK
  SDF rounded rectangle %s %s %s %s %s %s %s ['$sx', '$sy', '$sw', '$sh', '($sh / 2)', '$x', '$y']
  control_stop(STOP_OPTION=this script)
SDF rounded rectangle %s %s %s %s %s %s %s ['$sx', '$sy', '$sw', '$sh', '($sw / 2)', '$x', '$y']

DEFINE Scan Screen []
control_if(CONDITION=(#onmouse = 0))

  setvariableto(VARIABLE=#holdingshape, VALUE=0)
setvariableto(VARIABLE=#id, VALUE=0)
setvariableto(VARIABLE=#cx, VALUE=-240)
setvariableto(VARIABLE=#cy, VALUE=180)
control_repeat(TIMES=H)
SUBSTACK
  setvariableto(VARIABLE=#cx, VALUE=-240)
  control_repeat(TIMES=W)
  SUBSTACK
    setvariableto(VARIABLE=#is, VALUE=0)
    changevariableby(VARIABLE=#id, VALUE=1)
    control_repeat(TIMES=#shape num)

      changevariableby(VARIABLE=#is, VALUE=1)
      SDF id: %s position: %s %s ['#is', '#cx', '#cy']

      SUBSTACK
        control_if(CONDITION=(#onmouse = 0))

    replaceitemoflist(LIST=Distance Map, INDEX=#id, ITEM=#min distance)

  changevariableby(VARIABLE=#cy, VALUE=-1)

DEFINE Smooth union %s %s %s ["a","b","k"]
setvariableto(VARIABLE=#min distance, VALUE=((($a < $b) * $a) + (not(OPERAND=($a < $b)) * $b)))
control_if(CONDITION=($k = 0))
SUBSTACK
  control_stop(STOP_OPTION=this script)
setvariableto(VARIABLE=#h, VALUE=($k - abs(($a - $b))))
setvariableto(VARIABLE=#h, VALUE=((#h > 0) * #h))
setvariableto(VARIABLE=#min distance, VALUE=(#min distance - ((#h * #h) / (4 * $k))))

DEFINE Move shape %s %s %s ["id","x","y"]
replaceitemoflist(LIST=#shape x, INDEX=$id, ITEM=$x)

  SUBSTACK
    control_repeat(TIMES=16)

    changevariableby(VARIABLE=#color, VALUE=1792)

DEFINE Scan []
motion_sety(Y=180)
setvariableto(VARIABLE=#scan idx, VALUE=1)
control_repeat(TIMES=360)
SUBSTACK
  motion_setx(X=-240)
  control_repeat(TIMES=480)
  SUBSTACK
    control_if(CONDITION=not(OPERAND=sensing_touchingcolor(COLOR=#color)))
    SUBSTACK
      setvariableto(VARIABLE=#color, VALUE=(2 * Colour Map[#scan idx]))
      setvariableto(VARIABLE=i, VALUE=1)
      control_repeat_until(CONDITION=sensing_touchingcolor(COLOR=(#color + #colours[i])))
      SUBSTACK
        changevariableby(VARIABLE=i, VALUE=1)
      changevariableby(VARIABLE=#color, VALUE=#colours[i])
    replaceitemoflist(LIST=Colour Map, INDEX=#scan idx, ITEM=#color)
    changevariableby(VARIABLE=#scan idx, VALUE=1)
    motion_changexby(DX=1)
  motion_changeyby(DY=-1)

DEFINE Setup Colours List from %s depth %s ["color","depth"]

SUBSTACK
  control_repeat(TIMES=$depth)
  SUBSTACK
    control_repeat(TIMES=$depth)
    SUBSTACK
      control_if(CONDITION=not(OPERAND=listcontainsitem(LIST=#colours, ITEM=#color)))
      SUBSTACK
        addtolist(LIST=#colours, ITEM=#color)
